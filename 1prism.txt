@app.post("/upload")
async def upload_file(file: UploadFile = File(...)):

    content = await file.read()

    with open(file.filename, "wb") as f:
        f.write(content)

    parsed = parse_document(file.filename)

    return {
        "filename": file.filename,
        "parsed": parsed
    }


# --------------------------------------------------
# SIREN ALERT
# --------------------------------------------------

def siren_check(judge_output):

    alerts = []

    for item in judge_output:
        if item["risk"] == "high":
            alerts.append({
                "feature": item["feature_name"],
                "severity": "critical",
                "reason": "High risk feature detected"
            })

    return alerts


# --------------------------------------------------
# ANALYZE
# --------------------------------------------------

@app.post("/analyze", response_class=HTMLResponse)
async def analyze(request: Request, file: UploadFile = File(...)):

    try:

        # SAVE FILE
        content = await file.read()

        with open(file.filename, "wb") as f:
            f.write(content)

        # PARSE
        parsed = parse_document(file.filename)

        # AI PIPELINE
        pulse = PULSE()

        run_id = await create_run(parsed)

        result = await pulse.run(parsed)

        radar_output = result["radar"]
        judge_output = result["judge"]

        alerts = siren_check(judge_output)

        weaver = WEAVER()

        report = weaver.build_report(parsed, radar_output, judge_output)

        # SAVE REPORT TEMP (used for PDF)
        txt_file = f"{run_id}.txt"

        with open(txt_file, "w", encoding="utf-8") as f:
            f.write(report)

        return templates.TemplateResponse(
            "report.html",
            {
                "request": request,
                "parsed": parsed,
                "radar": radar_output,
                "judge": judge_output,
                "report": report,
                "alerts": alerts,
                "run_id": run_id
            }
        )

    except Exception as e:
        print("ANALYZE ERROR:", e)

        return HTMLResponse(
            content=f"<h1>PRISM ERROR</h1><p>{str(e)}</p>",
            status_code=500
        )

@app.get("/pdf/{run_id}")
async def download_pdf(run_id: str):

    import os
    from reportlab.platypus import SimpleDocTemplate, Paragraph, Spacer, Table, TableStyle
    from reportlab.lib.styles import getSampleStyleSheet, ParagraphStyle
    from reportlab.lib.pagesizes import A4
    from reportlab.lib import colors

    txt_file = f"{run_id}.txt"
    pdf_file = f"{run_id}.pdf"

    if not os.path.exists(txt_file):
        return HTMLResponse("<h1>Report not found</h1>", status_code=404)

    with open(txt_file, "r", encoding="utf-8") as f:
        content = f.readlines()

    # PDF setup
    pdf = SimpleDocTemplate(pdf_file, pagesize=A4)
    styles = getSampleStyleSheet()

    # Custom styles
    title_style = ParagraphStyle(
        "TitleStyle",
        parent=styles["Title"],
        fontSize=22,
        textColor=colors.darkblue,
        alignment=1,  # center
        spaceAfter=20
    )

    heading_style = ParagraphStyle(
        "HeadingStyle",
        parent=styles["Heading2"],
        textColor=colors.green,
        spaceAfter=10
    )

    normal_style = styles["Normal"]

    story = []

    # --------------------------
    # TITLE PAGE
    # --------------------------
    story.append(Paragraph("🚀 PRISM AI REPORT", title_style))
    story.append(Paragraph(f"Run ID: {run_id}", normal_style))
    story.append(Spacer(1, 20))

    # --------------------------
    # CONTENT
    # --------------------------
    story.append(Paragraph("📊 Analysis Report", heading_style))
    story.append(Spacer(1, 10))

    for line in content:

        line = line.strip()

        if not line:
            continue

        # Detect headings
        if line.startswith("#"):
            clean_line = line.replace("#", "").strip()
            story.append(Spacer(1, 10))
            story.append(Paragraph(f"<b>{clean_line}</b>", heading_style))

        # Bullet points
        elif line.startswith("-"):
            story.append(Paragraph(f"• {line[1:].strip()}", normal_style))

        # Normal text
        else:
            story.append(Paragraph(line, normal_style))

        story.append(Spacer(1, 4))

    # build PDF
    pdf.build(story)

    return FileResponse(
        path=pdf_file,
        filename="PRISM_AI_Report.pdf",
        media_type="application/pdf"
    )

@app.post("/atlas")
async def atlas_index(repo_url: str):

    atlas = ATLAS()
    atlas.clone_repo(repo_url)

    files = atlas.scan_files()

    return {
        "status": "indexed",
        "files": len(files),
        "file_list": files[:10]
    }
